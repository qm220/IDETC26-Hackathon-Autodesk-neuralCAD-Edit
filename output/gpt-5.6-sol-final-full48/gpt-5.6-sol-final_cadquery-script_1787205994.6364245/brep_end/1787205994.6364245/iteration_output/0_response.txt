def my_cad_function(args):
    import os
    import math

    input_file = os.path.expanduser(args["input_file"])
    imported = cq.importers.importStep(input_file)
    root = imported.val() if hasattr(imported, "val") else imported
    solids = list(root.Solids())
    if not solids:
        raise ValueError("The input STEP file contains no solids")

    def info(shape):
        bb = shape.BoundingBox()
        c = bb.center
        dx, dy, dz = bb.xlen, bb.ylen, bb.zlen
        diag = math.sqrt(dx * dx + dy * dy + dz * dz)
        box_volume = max(dx * dy * dz, 1.0e-9)
        volume = abs(shape.Volume())
        return {
            "shape": shape,
            "center": cq.Vector(c.x, c.y, c.z),
            "dims": (dx, dy, dz),
            "diag": diag,
            "volume": volume,
            "fill": volume / box_volume,
        }

    data = [info(s) for s in solids]
    main_index = max(range(len(data)), key=lambda i: data[i]["volume"])
    main_center = data[main_index]["center"]
    main_volume = data[main_index]["volume"]

    for i, d in enumerate(data):
        dc = d["center"] - main_center
        distance = dc.Length
        print(
            "Solid %d: center=(%.3f, %.3f, %.3f), dims=(%.3f, %.3f, %.3f), "
            "volume=%.3f, fill=%.4f, distance=%.3f"
            % (
                i,
                d["center"].x,
                d["center"].y,
                d["center"].z,
                d["dims"][0],
                d["dims"][1],
                d["dims"][2],
                d["volume"],
                d["fill"],
                distance,
            )
        )

    # The old molded plug is the compact, reasonably solid component farthest
    # from the main appliance. The flexible cable has a very low bounding-box
    # fill ratio and is intentionally excluded.
    candidates = []
    for i, d in enumerate(data):
        if i == main_index:
            continue
        distance = (d["center"] - main_center).Length
        if (
            d["fill"] > 0.08
            and d["volume"] < main_volume * 0.08
            and d["diag"] < 80.0
        ):
            candidates.append((distance, i))

    if candidates:
        plug_index = max(candidates)[1]
    else:
        # Conservative fallback: choose the compact component most remote from
        # the main body, while avoiding extremely sparse swept geometry.
        plug_index = max(
            (i for i in range(len(data)) if i != main_index),
            key=lambda i: (data[i]["center"] - main_center).Length
            * max(min(data[i]["fill"], 1.0), 0.02),
        )

    old_plug = data[plug_index]["shape"]
    plug_center = data[plug_index]["center"]
    print("Selected old plug solid index:", plug_index)

    # Find the preserved flexible cord. It is a remote, elongated component

    attach = cq.Vector(

                length,
                rib_start,

        )

    pin_shapes = []

    pin_radius = 2.0
    sleeve_projection = 10.0
    conductive_straight = 8.3
    tip_length = 0.7

    for side in (-1.0, 1.0):
        pin_center = nose + width_dir.multiply(side * pin_spacing / 2.0)
        sleeve_start = pin_center - insertion_axis.multiply(1.0)
        sleeve = cq.Solid.makeCylinder(
            pin_radius,
            sleeve_projection + 1.0,
            sleeve_start,
            insertion_axis,
        )
        metal_start = pin_center + insertion_axis.multiply(sleeve_projection)
        metal = cq.Solid.makeCylinder(
            pin_radius,
            conductive_straight,
            metal_start,
            insertion_axis,
        )

            pin_radius,
            1.65,
            tip_length,
            tip_start,
            insertion_axis,

    replacement_parts = [body] + boot_shapes + pin_shapes

        "34 mm body width, 14 mm body thickness, two 4 mm round pins, "
        "19 mm pin spacing and 19 mm projection."
