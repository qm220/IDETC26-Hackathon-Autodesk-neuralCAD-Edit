    hinge_boss_r = 2.15
    hinge_pin_r = 1.0
    hinge_bore_r = 1.10

    latch_boss_r = 1.65
    latch_pin_r = 0.80
    latch_bore_r = 0.92

    def y_cylinder(radius, y0, y1, x, z):
        solid = cq.Solid.makeCylinder(
            radius,
            y1 - y0,
            cq.Vector(x, y0, z),
            cq.Vector(0, 1, 0)
        )
        return cq.Workplane(obj=solid)

    # the independently moving gate.
    hinge_ear_neg = y_cylinder(

    body = (
