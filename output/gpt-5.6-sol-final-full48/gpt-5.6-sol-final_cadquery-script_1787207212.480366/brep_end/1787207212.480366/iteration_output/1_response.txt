    # Load the source model as required. The replacement body below is rebuilt

    input_file = os.path.expanduser(args['input_file'])

    x0, x_joint, x1 = 0.0, 100.0, 300.0

        x_joint - x0,
        big_y1 - big_y0,
        z1 - z0,
        cq.Vector(x0, big_y0, z0)
    )
    stem = cq.Solid.makeBox(
        x1 - x_joint,
        stem_y1 - stem_y0,
        z1 - z0,
        cq.Vector(x_joint, stem_y0, z0)
    )
    body = big.fuse(stem)

    def vertical_edges(shape):
        result = []
        for edge in shape.Edges():
            eb = edge.BoundingBox()
            if (edge.geomType() == 'LINE' and
                eb.zlen > 0.90 * (z1 - z0) and
                eb.xlen < 1.0e-5 and eb.ylen < 1.0e-5):
                result.append(edge)
        return result

    # R20 at the two symmetric narrow-to-large-body shoulder corners.
    shoulders = []

        c = edge.Center()
        if (abs(c.x - x_joint) < 1.0e-4 and

             abs(c.y - stem_y1) < 1.0e-4)):
            shoulders.append(edge)
    if shoulders:
        body = body.fillet(20.0, shoulders)

    # R5 on all remaining vertical exterior corners.
    exterior_vertical = vertical_edges(body)
    if exterior_vertical:

    # R5 on the upper and lower exterior perimeter edges. These are selected

    horizontal_perimeter = []
    for edge in body.Edges():

    if horizontal_perimeter:
