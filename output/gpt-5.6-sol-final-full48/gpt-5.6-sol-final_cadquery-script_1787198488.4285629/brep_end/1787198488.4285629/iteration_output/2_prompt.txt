def my_cad_function(args):
    import os
    import cadquery as cq

    input_file = os.path.expanduser(args["input_file"])
    imported = cq.importers.importStep(input_file)
    root = imported.val()
    solids = list(root.Solids())

    if len(solids) < 2:
        raise ValueError("Expected separate housing and scroll-wheel solids")

    # Identify the housing by its dominant volume and preserve every other solid.
    solids_by_volume = sorted(solids, key=lambda solid: solid.Volume(), reverse=True)
    housing = solids_by_volume[0]
    preserved_solids = solids_by_volume[1:]

    faces = list(housing.Faces())
    if len(faces) < 7:
        raise ValueError("Housing does not contain the expected wheel-slot topology")

    # Source topology: faces 0-3 are the planar slot walls and face 6 is the
    # surrounding upper housing deck.
    well_walls = [faces[index] for index in (0, 1, 2, 3)]
    top_deck = faces[6]

    def same_shape(shape_a, shape_b):
        try:
            return shape_a.wrapped.IsSame(shape_b.wrapped)
        except Exception:
            return shape_a.isSame(shape_b)

    # Select the four edges shared by the upper deck and the wheel-well walls.
    # Lower wall/floor edges and all edges belonging to the wheel are excluded.
    deck_edges = list(top_deck.Edges())
    slot_rim_edges = []
    for wall in well_walls:
        for wall_edge in wall.Edges():
            if any(same_shape(wall_edge, deck_edge) for deck_edge in deck_edges):
                if not any(same_shape(wall_edge, edge) for edge in slot_rim_edges):
                    slot_rim_edges.append(wall_edge)

    print("Input solids:", len(solids))
    print("Housing faces:", len(faces))
    print("Selected upper wheel-slot rim edges:", len(slot_rim_edges))
    for index, edge in enumerate(slot_rim_edges):
        center = edge.Center()
        print(
            "  edge %d: length=%.6f center=(%.6f, %.6f, %.6f)"
            % (index, edge.Length(), center.x, center.y, center.z)
        )

    if len(slot_rim_edges) != 4:
        raise ValueError(
            "Expected four exposed wheel-slot perimeter edges, found %d"
            % len(slot_rim_edges)
        )

    # Apply the fillet through Workplane.fillet; the imported Solid itself does
    # not expose makeFillet in this CadQuery version.
    housing_workplane = cq.Workplane(obj=housing)
    filleted_housing = housing_workplane.newObject(slot_rim_edges).fillet(2.0).val()

    if filleted_housing is None or not filleted_housing.isValid():
        raise ValueError("The 2 mm wheel-slot fillet produced an invalid housing")

    result = cq.Compound.makeCompound([filleted_housing] + preserved_solids)
    if not result.isValid():
        raise ValueError("The resulting housing and preserved wheel compound is invalid")

    print("Applied a continuous 2.0 mm fillet to the four upper slot-rim edges")
    print("Result valid:", result.isValid())
    print("Result solids:", len(result.Solids()))
    return result
