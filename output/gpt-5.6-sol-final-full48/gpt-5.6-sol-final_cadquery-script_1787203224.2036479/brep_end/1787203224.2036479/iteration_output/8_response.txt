        raise ValueError("The input STEP file contains no solids")

    blade_solids = []
    preserved_solids = []
    for solid in source_solids:
        bb = solid.BoundingBox()
        if max(bb.xlen, bb.zlen) > 80.0:
            blade_solids.append(solid)
        else:
            preserved_solids.append(solid)

    if len(blade_solids) != 2:
        raise ValueError("Expected exactly two original rotor-blade solids")

    edge_radius = 1.0

    def round_long_edges(blade):
        candidates = []
        for edge in blade.Edges():
            try:
                if edge.geomType() == "LINE" and edge.Length() > 30.0:
                    candidates.append(edge)
            except Exception:
                pass

        if not candidates:
            return blade, 0

        try:
            return blade.fillet(edge_radius, candidates), len(candidates)
        except Exception:
            result = blade
            applied = 0
            for original_edge in candidates:
                center = original_edge.Center()
                best = None
                best_distance = 1.0e100
                for current_edge in result.Edges():
                    try:
                        if current_edge.geomType() != "LINE" or current_edge.Length() <= 30.0:
                            continue
                        c = current_edge.Center()
                        d = ((c.x - center.x) ** 2 +
                             (c.y - center.y) ** 2 +
                             (c.z - center.z) ** 2)
                        if d < best_distance:
                            best_distance = d
                            best = current_edge
                    except Exception:
                        pass
                if best is not None:
                    try:
                        result = result.fillet(edge_radius, [best])
                        applied += 1
                    except Exception:
                        pass
            return result, applied

    rounded_blades = []
    rounded_counts = []
    for blade in blade_solids:
        rounded, count = round_long_edges(blade)
        rounded_blades.append(rounded)
        rounded_counts.append(count)

    inner_radius = 14.0
    transition_radius = 22.0

        thin_half = central_thickness / 2.0
        full_half = max(full_half_thickness, thin_half + 0.05)
        profile = (
