                it = faces.cbegin()

    # --- Apply chamfer: try all-at-once; if it fails, do sequential edge-by-edge ---
    def find_matching_edge(shape, center_vec, r, r_tol=0.02, c_tol=0.5):
        """Find a full-circle edge in 'shape' matching radius and center."""
        best = None
        best_d = 1e99
