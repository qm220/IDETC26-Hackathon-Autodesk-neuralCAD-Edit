def my_cad_function(args):
    import os
    import cadquery as cq

    t = 1.5  # rib thickness (mm)

    if "input_file" not in args:
        print("No input_file provided; cannot edit model.")
        return None

    input_file = os.path.expanduser(args["input_file"])
    model = cq.importers.importStep(input_file)

    # Underlying OCC shape
    shp = model.val() if hasattr(model, "val") else model

    bbox = shp.BoundingBox()
    cx, cy, cz = bbox.center.x, bbox.center.y, bbox.center.z
    xlen, ylen, zlen = bbox.xlen, bbox.ylen, bbox.zlen

    print("Loaded STEP:", input_file)
    print(f"Valid: {shp.isValid()}")
    print(f"BBOX center=({cx:.3f},{cy:.3f},{cz:.3f})")
    print(f"BBOX lens x={xlen:.3f} y={ylen:.3f} z={zlen:.3f}")

    # Try to identify a large horizontal base face and a large vertical wall face
    base_face = None
    wall_face = None
    base_area = -1.0
    wall_area = -1.0

    faces = shp.Faces()
    print(f"Faces: {len(faces)}")

    for f in faces:
        try:
            if hasattr(f, "geomType") and f.geomType() != "PLANE":
                continue
            # normalAt() in CadQuery typically returns normal at center when called without params
            n = f.normalAt()
            a = f.Area()
            c = f.Center()
        except Exception:
            continue

        # base: mostly horizontal (normal along +/-Z)
        if abs(n.z) > 0.85 and a > base_area:
            base_face, base_area = (f, a)

        # wall: mostly vertical (normal along +/-X or +/-Y)
        if abs(n.z) < 0.25 and max(abs(n.x), abs(n.y)) > 0.85 and a > wall_area:
            wall_face, wall_area = (f, a)

    if base_face is None or wall_face is None:
        print("WARNING: Could not robustly detect base/wall planar faces. Will fall back to bbox-based rib placement.")

    # Decide wall axis (X or Y) from wall normal
    wall_axis = None
    wall_center = None
    if wall_face is not None:
        wn = wall_face.normalAt()
        wc = wall_face.Center()
        wall_center = wc
        wall_axis = "X" if abs(wn.x) >= abs(wn.y) else "Y"
        print(f"Detected wall axis: {wall_axis}, wall center=({wc.x:.3f},{wc.y:.3f},{wc.z:.3f}), wall area={wall_area:.3f}")
    else:
        # fallback: choose axis with smaller bbox length as thickness direction -> wall likely perpendicular to largest span
        wall_axis = "X" if xlen >= ylen else "Y"

    # Decide base side (zmin or zmax) from base face center
    if base_face is not None:
        bc = base_face.Center()
        base_side = "zmin" if abs(bc.z - bbox.zmin) < abs(bc.z - bbox.zmax) else "zmax"
        print(f"Detected base side: {base_side}, base center z={bc.z:.3f}, base area={base_area:.3f}")
    else:
        base_side = "zmin"
        print("Fallback base side: zmin")

    # Choose wall plane location using bbox side closest to wall face center
    if wall_axis == "X":
        if wall_center is not None:

        else:
            wall_at_min = True
        wall_x = bbox.xmin if wall_at_min else bbox.xmax
        sx = 1.0 if wall_at_min else -1.0  # interior direction in X

        width_axis = "Y"

        base_z = bbox.zmin if base_side == "zmin" else bbox.zmax
        sz = 1.0 if base_side == "zmin" else -1.0  # interior direction in Z

        # Rib size heuristics
        Lx = max(6.0, 0.30 * xlen)
        Lz = max(6.0, 0.30 * zlen)
        L = min(Lx, Lz)

        off = 0.2

        x1 = x0 + sx * Lx
        z1 = z0 + sz * Lz

        rib = (

        if wall_center is not None:

        width_center = cx

        base_z = bbox.zmin if base_side == "zmin" else bbox.zmax
        sz = 1.0 if base_side == "zmin" else -1.0

        Ly = max(6.0, 0.30 * ylen)

        rib = (

            .translate((width_center, 0, 0))
        )

    # Union rib to model
    try:
        result = cq.Workplane(obj=shp).union(rib)

        # Fallback: union as shapes
        result = model.union(rib)

    print(f"Added rib thickness={t}mm (extruded along {width_axis})")
    return result
