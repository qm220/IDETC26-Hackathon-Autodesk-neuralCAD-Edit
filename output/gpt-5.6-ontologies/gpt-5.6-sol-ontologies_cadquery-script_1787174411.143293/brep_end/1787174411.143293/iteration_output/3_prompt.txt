def my_cad_function(args):
    import os
    import math
    import cadquery as cq
    from OCP.gp import gp_Pnt, gp_Dir, gp_Pln, gp_Trsf
    from OCP.BRepBuilderAPI import BRepBuilderAPI_Transform
    from OCP.BRepOffsetAPI import BRepOffsetAPI_DraftAngle
    from OCP.BRepFilletAPI import BRepFilletAPI_MakeFillet

    input_file = os.path.expanduser(args["input_file"])
    imported = cq.importers.importStep(input_file)
    source = imported.val() if hasattr(imported, "val") else imported

    if not source.isValid() or len(source.Solids()) != 1:
        raise ValueError("Expected one valid source solid")

    # Scale only the pre-existing part. Dimensions stated for subsequently
    # added rounds and cylinders are treated as final absolute millimetres.
    scale_transform = gp_Trsf()
    scale_transform.SetScale(gp_Pnt(0.0, 0.0, 0.0), 10.0)
    scale_op = BRepBuilderAPI_Transform(source.wrapped, scale_transform, True)
    scale_op.Build()
    edited = cq.Shape.cast(scale_op.Shape())

    if not edited.isValid():
        raise RuntimeError("The scaled part is invalid")

    bb = edited.BoundingBox()
    bottom_z = bb.zmin
    center_x = 0.5 * (bb.xmin + bb.xmax)
    center_y = 0.5 * (bb.ymin + bb.ymax)
    print(
        "Scaled bbox: "
        f"x=({bb.xmin:.4f},{bb.xmax:.4f}), "
        f"y=({bb.ymin:.4f},{bb.ymax:.4f}), "
        f"z=({bb.zmin:.4f},{bb.zmax:.4f})"
    )

    # Draft every planar vertical wall of the existing part by two degrees.
    # The preserved bottom plane is the neutral/hinge plane.
    vertical_faces = []
    for face in edited.Faces():
        if face.geomType() != "PLANE":
            continue
        try:
            normal = face.normalAt(face.Center())
        except Exception:
            continue
        if abs(normal.z) < 1.0e-6 and face.BoundingBox().zlen > 1.0e-5:
            vertical_faces.append(face)

    print(f"Vertical planar faces selected for draft: {len(vertical_faces)}")
    neutral_plane = gp_Pln(
        gp_Pnt(center_x, center_y, bottom_z),
        gp_Dir(0.0, 0.0, 1.0)
    )
    pull_direction = gp_Dir(0.0, 0.0, 1.0)
    draft_angle = math.radians(2.0)

    drafted = None
    draft_error = None
    for flag in (True, False):
        try:
            draft_op = BRepOffsetAPI_DraftAngle(edited.wrapped)
            for face in vertical_faces:
                draft_op.Add(
                    face.wrapped,
                    pull_direction,
                    draft_angle,
                    neutral_plane,
                    flag
                )
            draft_op.Build()
            candidate = cq.Shape.cast(draft_op.Shape())
            if candidate.isValid() and len(candidate.Solids()) == 1:
                drafted = candidate
                print(f"2 degree draft completed with flag={flag}")
                break
        except Exception as exc:
            draft_error = exc
            print(f"Draft attempt flag={flag} failed: {exc}")

    if drafted is None:
        raise RuntimeError(f"Unable to apply requested draft: {draft_error}")
    edited = drafted

    bottom_tol = 1.0e-4

    def lies_on_bottom(edge):
        edge_bb = edge.BoundingBox()
        return (
            edge_bb.zlen <= bottom_tol
            and abs(edge_bb.zmin - bottom_z) <= bottom_tol
        )

    # On this symmetric cradle, the underside-relief edges lie inside the
    # inset cavity footprint. Exterior edges reach the outer X or Y envelope.
    # Use dimensions relative to the scaled bounding box rather than relying
    # on CadQuery topology indices.
    half_x = 0.5 * bb.xlen
    half_y = 0.5 * bb.ylen
    inner_x_limit = 0.92 * half_x
    inner_y_limit = 0.9733333333 * half_y

    def is_inner_edge(edge):
        edge_bb = edge.BoundingBox()
        x_extent = max(
            abs(edge_bb.xmin - center_x),
            abs(edge_bb.xmax - center_x)
        )
        y_extent = max(
            abs(edge_bb.ymin - center_y),
            abs(edge_bb.ymax - center_y)
        )
        return x_extent < inner_x_limit and y_extent < inner_y_limit

    def apply_fillet(shape, radius, edges, label):
        if not edges:
            print(f"No edges selected for {label}")
            return shape

        fillet_op = BRepFilletAPI_MakeFillet(shape.wrapped)
        for edge in edges:
            fillet_op.Add(radius, edge.wrapped)
        fillet_op.Build()

        if not fillet_op.IsDone():
            raise RuntimeError(f"OpenCascade could not build {label}")

        result = cq.Shape.cast(fillet_op.Shape())
        if not result.isValid() or len(result.Solids()) != 1:
            raise RuntimeError(f"{label} produced an invalid result")
        return result

    # Apply R1 simultaneously to the non-bottom concave/cavity edge network.
    inner_edges = [
        edge for edge in edited.Edges()
        if not lies_on_bottom(edge) and is_inner_edge(edge)
    ]
    print(f"Inner non-bottom edges selected for R1: {len(inner_edges)}")
    edited = apply_fillet(edited, 1.0, inner_edges, "R1 inner rounds")

    # Recompute topology after R1 and apply R3 to edges at the exterior
    # envelope. Bottom-plane edges remain explicitly excluded.
    outer_edges = []
    for edge in edited.Edges():
        if lies_on_bottom(edge):
            continue
        edge_bb = edge.BoundingBox()
        x_extent = max(
            abs(edge_bb.xmin - center_x),
            abs(edge_bb.xmax - center_x)
        )
        y_extent = max(
            abs(edge_bb.ymin - center_y),
            abs(edge_bb.ymax - center_y)
        )
        if x_extent >= inner_x_limit or y_extent >= inner_y_limit:
            outer_edges.append(edge)

    print(f"Outer non-bottom edges selected for R3: {len(outer_edges)}")
    edited = apply_fillet(edited, 3.0, outer_edges, "R3 outer rounds")

    # Add a pair of coaxially hollow D6 cylinders. Their axes are 30 mm apart
    # along the long Y direction and their midpoint is the part center.
    outer_radius = 3.0
    hole_radius = 1.5
    cylinder_y_positions = (center_y - 15.0, center_y + 15.0)

    # The original cradle underside is a cylindrical wall. After 10x scaling,
    # its axis-height and underside radius are 42.15 and 44.15 mm. Evaluate the
    # wall at each cylinder axis. A 1 mm overlap robustly joins each boss to the
    # wall, while cutting the hole before union makes the hole pass through the
    # full boss but remain blind at the existing cradle wall.
    shell_center_z = 42.15
    underside_radius = 44.15
    axis = cq.Vector(0.0, 0.0, 1.0)

    for y_pos in cylinder_y_positions:
        y_offset = y_pos - center_y
        radial_term = underside_radius * underside_radius - y_offset * y_offset
        if radial_term <= 0.0:
            raise RuntimeError("Cylinder position lies outside the cradle wall")

        roof_z = shell_center_z - math.sqrt(radial_term)
        boss_height = roof_z - bottom_z + 1.0
        hole_height = roof_z - bottom_z + 0.02
        if boss_height <= 1.0 or hole_height <= 0.0:
            raise RuntimeError("Calculated cylindrical-feature height is invalid")

        base_point = cq.Vector(center_x, y_pos, bottom_z)
        outer_cylinder = cq.Solid.makeCylinder(
            outer_radius, boss_height, base_point, axis
        )
        inner_cylinder = cq.Solid.makeCylinder(
            hole_radius, hole_height, base_point, axis
        )
        hollow_boss = outer_cylinder.cut(inner_cylinder)
        edited = edited.fuse(hollow_boss)

        print(
            f"Added hollow cylinder axis=({center_x:.4f},{y_pos:.4f}), "
            f"OD=6, ID=3, start_z={bottom_z:.4f}, wall_z={roof_z:.4f}"
        )

    try:
        edited = edited.clean()
    except Exception as exc:
        print(f"Final clean skipped: {exc}")

    if not edited.isValid():
        raise RuntimeError("Final edited part is invalid")
    if len(edited.Solids()) != 1:
        raise RuntimeError(
            f"Expected one integrated final solid, found {len(edited.Solids())}"
        )

    final_bb = edited.BoundingBox()
    print(
        f"Final solid count={len(edited.Solids())}, valid={edited.isValid()}, "
        f"bbox size=({final_bb.xlen:.4f}, "
        f"{final_bb.ylen:.4f}, {final_bb.zlen:.4f})"
    )

    return cq.Workplane("XY").newObject([edited])
