                        break

                    pass
        print("SOLID %d volume=%.3f center=(%.3f, %.3f, %.3f) bbox=x[%.3f,%.3f] y[%.3f,%.3f] z[%.3f,%.3f] faces=%s" %

               global_face_ids))

    return model
