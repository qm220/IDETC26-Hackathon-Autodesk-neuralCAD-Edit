def my_cad_function(args):
    import os
    import cadquery as cq

    input_file = os.path.expanduser(args["input_file"])
    model = cq.importers.importStep(input_file)
    original = model.val()

    # The vertical elongated relief consists of aligned blind pockets on the
    # front and back sides. Remove the intervening web to make one continuous
    # through-slot while preserving its existing 18 mm width and 48 mm length.
    # Slot center: (113, 55), end-center spacing: 30 mm, end radius: 9 mm.
    z0 = -61.0
    cut_depth = 62.0

    center_section = cq.Workplane("XY").box(
        18.0, 30.0, cut_depth,
        centered=(True, True, False)
    ).translate((113.0, 55.0, z0))

    lower_end = cq.Workplane("XY", origin=(113.0, 40.0, z0)).circle(9.0).extrude(cut_depth)
    upper_end = cq.Workplane("XY", origin=(113.0, 70.0, z0)).circle(9.0).extrude(cut_depth)

    through_slot = center_section.union(lower_end).union(upper_end)
    result = cq.Workplane(obj=original).cut(through_slot)

    final_shape = result.val()
    print("ORIGINAL VOLUME: %.6f" % original.Volume())
    print("FINAL VOLUME: %.6f" % final_shape.Volume())
    print("VOLUME REMOVED: %.6f" % (original.Volume() - final_shape.Volume()))
    print("FINAL VALID:", final_shape.isValid())
    print("VERTICAL SLOT EXTENT: through complete body, z=-60 to 0")

    return result
