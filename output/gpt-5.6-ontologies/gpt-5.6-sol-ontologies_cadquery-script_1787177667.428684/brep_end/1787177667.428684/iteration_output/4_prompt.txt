def my_cad_function(args):
    input_file = os.path.expanduser(args["input_file"])
    model = cq.importers.importStep(input_file)
    source_shape = model.val()
    source_bb = source_shape.BoundingBox()

    # Dimensions in millimetres: 200 mm wide, 100 mm high,
    # 30 mm deep, with 10 mm corner radii.
    opening_width = 200.0
    opening_height = 100.0
    opening_depth = 30.0
    corner_radius = 10.0

    # The rendered orthographic views show that physical width, height, and
    # depth correspond to model X, Y, and Z respectively. The smooth back is
    # the minimum-Z exterior, while minimum Y is the physical bottom.
    center_x = 0.5 * (source_bb.xmin + source_bb.xmax)
    side_clearance = 0.5 * ((source_bb.xmax - source_bb.xmin) - opening_width)

    # Match the bottom clearance approximately to the two side clearances.
    bottom_clearance = side_clearance
    xmin = center_x - opening_width / 2.0
    ymin = source_bb.ymin + bottom_clearance

    rear_z = source_bb.zmin
    outer_overshoot = 4.0
    cutter_z0 = rear_z - outer_overshoot
    cutter_length = opening_depth + outer_overshoot

    # Build a single rounded-rectangular prism extending inward from the
    # minimum-Z rear surface. It consists of two overlapping central boxes
    # and four radius-10 corner cylinders.
    cutter_parts = [
        cq.Solid.makeBox(
            opening_width - 2.0 * corner_radius,
            opening_height,
            cutter_length,
            cq.Vector(xmin + corner_radius, ymin, cutter_z0)
        ),
        cq.Solid.makeBox(
            opening_width,
            opening_height - 2.0 * corner_radius,
            cutter_length,
            cq.Vector(xmin, ymin + corner_radius, cutter_z0)
        )
    ]

    for x in (xmin + corner_radius,
              xmin + opening_width - corner_radius):
        for y in (ymin + corner_radius,
                  ymin + opening_height - corner_radius):
            cutter_parts.append(
                cq.Solid.makeCylinder(
                    corner_radius,
                    cutter_length,
                    cq.Vector(x, y, cutter_z0),
                    cq.Vector(0, 0, 1)
                )
            )

    cutter = cutter_parts[0]
    for part in cutter_parts[1:]:
        cutter = cutter.fuse(part)

    # Cut the imported assembly in one operation. Validate that material was
    # actually removed; otherwise use a robust per-solid fallback.
    source_volume = sum(s.Volume() for s in model.solids().vals())
    use_fallback = False

    try:
        result_shape = source_shape.cut(cutter)
        if result_shape.wrapped.IsNull() or not result_shape.isValid():
            use_fallback = True
        else:
            result_volume = sum(s.Volume() for s in result_shape.Solids())
            if source_volume - result_volume < 1.0:
                use_fallback = True
    except Exception as direct_error:
        print("DIRECT CUT ERROR:", direct_error)
        use_fallback = True

    if use_fallback:
        edited_shapes = []
        cut_count = 0
        preserved_count = 0
        failed_count = 0

        x1 = xmin + opening_width
        y1 = ymin + opening_height
        z1 = rear_z + opening_depth

        for solid in model.solids().vals():
            bb = solid.BoundingBox()
            overlaps = not (
                bb.xmax <= xmin or bb.xmin >= x1 or
                bb.ymax <= ymin or bb.ymin >= y1 or
                bb.zmax <= cutter_z0 or bb.zmin >= z1
            )

            if not overlaps:
                edited_shapes.append(solid)
                preserved_count += 1
                continue

            try:
                edited = solid.cut(cutter)
                if not edited.wrapped.IsNull():
                    edited_shapes.append(edited)
                cut_count += 1
            except Exception as solid_error:
                print("SOLID CUT WARNING:", solid_error)
                edited_shapes.append(solid)
                failed_count += 1

        if not edited_shapes:
            raise ValueError("Opening operation removed all model geometry")

        result_shape = cq.Compound.makeCompound(edited_shapes)
        boolean_mode = "per-solid fallback"
        print("FALLBACK COUNTS: cut=%d preserved=%d failed=%d" % (
            cut_count, preserved_count, failed_count))
    else:
        boolean_mode = "single compound cut"

    final_volume = sum(s.Volume() for s in result_shape.Solids())
    removed_volume = source_volume - final_volume
    result_bb = result_shape.BoundingBox()

    print("RESULT VALID:", result_shape.isValid())
    print("BOOLEAN MODE:", boolean_mode)
    print("SOURCE BBOX: x=[%.3f, %.3f] y=[%.3f, %.3f] z=[%.3f, %.3f]" % (
        source_bb.xmin, source_bb.xmax,
        source_bb.ymin, source_bb.ymax,
        source_bb.zmin, source_bb.zmax))
    print("RESULT BBOX: x=[%.3f, %.3f] y=[%.3f, %.3f] z=[%.3f, %.3f]" % (
        result_bb.xmin, result_bb.xmax,
        result_bb.ymin, result_bb.ymax,
        result_bb.zmin, result_bb.zmax))
    print("REMOVED VOLUME: %.3f mm^3" % removed_volume)
    print("REAR OPENING: width=%.1f height=%.1f depth=%.1f radius=%.1f" % (
        opening_width, opening_height, opening_depth, corner_radius))
    print("OPENING BOUNDS: x=[%.3f, %.3f] y=[%.3f, %.3f] z=[%.3f, %.3f]" % (
        xmin, xmin + opening_width,
        ymin, ymin + opening_height,
        rear_z, rear_z + opening_depth))
    print("APPROXIMATE SIDE/BOTTOM CLEARANCE: %.3f mm" % side_clearance)

    return cq.Workplane("XY").newObject([result_shape])
