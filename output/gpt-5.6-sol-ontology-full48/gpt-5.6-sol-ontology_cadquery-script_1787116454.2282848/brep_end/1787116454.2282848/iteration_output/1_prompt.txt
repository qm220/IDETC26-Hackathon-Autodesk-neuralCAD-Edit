            c = f.Center()

                gt = f.geomType()
            except Exception:
                gt = "UNKNOWN"
            print(
                "FACE %d: type=%s center=(%.4f, %.4f, %.4f) "
                "bbox=[x %.4f..%.4f, y %.4f..%.4f, z %.4f..%.4f] area=%.4f"
                % (idx, gt, c.x, c.y, c.z, bb.xmin, bb.xmax,

            )

        raise ValueError("The imported STEP contains no solids")
