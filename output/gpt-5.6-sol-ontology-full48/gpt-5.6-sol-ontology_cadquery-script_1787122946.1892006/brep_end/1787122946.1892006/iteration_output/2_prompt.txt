def my_cad_function(args):
    import os
    import cadquery as cq

    input_file = os.path.expanduser(args["input_file"])
    imported = cq.importers.importStep(input_file)
    model = imported.val() if hasattr(imported, "val") else imported

    faces = model.Faces()
    solids = model.Solids()
    bbox = model.BoundingBox()
    print(f"Loaded model: valid={model.isValid()}, solids={len(solids)}, faces={len(faces)}")
    print(
        f"Overall bounds: x=({bbox.xmin:.3f},{bbox.xmax:.3f}), "
        f"y=({bbox.ymin:.3f},{bbox.ymax:.3f}), "
        f"z=({bbox.zmin:.3f},{bbox.zmax:.3f})"
    )

    # Bind the planning-stage face indices to the actual imported topology.
    for index in (14, 17, 71, 74):
        if index < len(faces):
            face = faces[index]
            fb = face.BoundingBox()
            fc = face.Center()
            try:
                normal = face.normalAt()
                normal_text = f"({normal.x:.4f},{normal.y:.4f},{normal.z:.4f})"
            except Exception:
                normal_text = "unavailable"
            print(
                f"FACE {index}: center=({fc.x:.3f},{fc.y:.3f},{fc.z:.3f}), "
                f"bounds=x({fb.xmin:.3f},{fb.xmax:.3f}) "
                f"y({fb.ymin:.3f},{fb.ymax:.3f}) "
                f"z({fb.zmin:.3f},{fb.zmax:.3f}), "
                f"normal={normal_text}, area={face.Area():.3f}"
            )

    if len(faces) <= 14:
        raise ValueError("The imported model does not contain grounded FACE 14")

    # FACE 14 is the exterior face of the rear closure panel. If topology order
    # changes, recover it using its measured planar envelope and rearward Y.
    rear_face = faces[14]
    rear_bounds = rear_face.BoundingBox()
    rear_index = 14

    def is_rear_panel_candidate(face):
        b = face.BoundingBox()
        return b.xlen > 280.0 and b.zlen > 330.0 and b.ylen < 0.01 and b.ymax < 1.0

    if not is_rear_panel_candidate(rear_face):
        candidates = []
        for i, face in enumerate(faces):
            if is_rear_panel_candidate(face):
                b = face.BoundingBox()
                score = abs(b.xlen - 292.0) + abs(b.zlen - 344.0) + abs(b.ymin + 1.0)
                candidates.append((score, i, face))
        if not candidates:
            raise ValueError("Could not locate the large planar rear closure face")
        candidates.sort(key=lambda item: item[0])
        _, rear_index, rear_face = candidates[0]
        rear_bounds = rear_face.BoundingBox()
        print(f"Recovered rear closure as inspected FACE {rear_index}")

    opening_width = 200.0
    opening_height = 100.0
    corner_radius = 10.0
    cut_depth = 30.0

    panel_x_center = 0.5 * (rear_bounds.xmin + rear_bounds.xmax)
    panel_bottom = rear_bounds.zmin
    rear_y = 0.5 * (rear_bounds.ymin + rear_bounds.ymax)

    # The measured panel is 292 mm wide. Centering a 200 mm opening leaves
    # 46 mm at each side; use the same nominal clearance above its bottom.
    side_clearance = 0.5 * (rear_bounds.xlen - opening_width)
    opening_bottom = panel_bottom + side_clearance
    opening_z_center = opening_bottom + 0.5 * opening_height

    if side_clearance <= 0.0 or opening_bottom <= panel_bottom:
        raise ValueError("The requested opening does not fit within the rear panel")

    print(
        f"Rear opening: center=({panel_x_center:.3f},{rear_y:.3f},{opening_z_center:.3f}), "
        f"size={opening_width:.1f}x{opening_height:.1f} mm, "
        f"radius={corner_radius:.1f} mm, depth={cut_depth:.1f} mm, "
        f"side/bottom clearance={side_clearance:.3f} mm"
    )

    # XZ sketch plane, extruding inward from the semantic back toward +Y.
    # Start 0.1 mm outside the face to prevent a coincident residual skin.

    extrusion = cut_depth + 0.1

    # Build an exact rounded rectangle robustly from two overlapping prisms

    cutter_parts = [

          .rect(opening_width - 2.0 * corner_radius, opening_height)
          .extrude(extrusion)
          .val(),
        cq.Workplane(cut_plane)
          .rect(opening_width, opening_height - 2.0 * corner_radius)
          .extrude(extrusion)
          .val()
    ]

    corner_x = 0.5 * opening_width - corner_radius
    corner_z_local = 0.5 * opening_height - corner_radius
    for local_x in (-corner_x, corner_x):
        for local_z in (-corner_z_local, corner_z_local):

                cq.Workplane(cut_plane)
                .center(local_x, local_z)
                .circle(corner_radius)
                .extrude(extrusion)
                .val()

            cutter_parts.append(corner_tool)

    # within the specified 30 mm cavity depth.
    edited_shapes = []

    total_removed = 0.0

        edited = original_solid
        try:

            after_volume = edited.Volume()
            removed = before_volume - after_volume
        except Exception as exc:

        if removed > 1.0e-5:

            print(f"Cut R01 SOLID {solid_index}; removed volume={removed:.3f} mm^3")

        resulting_solids = edited.Solids()
        if resulting_solids:
            edited_shapes.extend(resulting_solids)
        else:

        raise ValueError("The edited assembly is not a valid B-rep")

    result_bbox = result.BoundingBox()
    print(f"Changed R01 solids: {changed_indices}")
    print(f"Total removed volume: {total_removed:.3f} mm^3")
    print(
        f"Result: valid={result.isValid()}, solids={len(result.Solids())}, "
        f"faces={len(result.Faces())}"
    )
    print(
        f"Result bounds: x=({result_bbox.xmin:.3f},{result_bbox.xmax:.3f}), "
        f"y=({result_bbox.ymin:.3f},{result_bbox.ymax:.3f}), "
        f"z=({result_bbox.zmin:.3f},{result_bbox.zmax:.3f})"
    )
    return result
