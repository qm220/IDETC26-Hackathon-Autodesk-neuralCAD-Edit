def my_cad_function(args):
    import os
    import cadquery as cq

    input_file = os.path.expanduser(args['input_file'])
    model = cq.importers.importStep(input_file)
    source_shape = model.val()
    bbox = source_shape.BoundingBox()

    print('SOURCE VALID:', source_shape.isValid())
    print('SOURCE SOLIDS:', len(source_shape.Solids()), 'FACES:', len(source_shape.Faces()))
    print('SOURCE BBOX: x=[%.2f, %.2f] y=[%.2f, %.2f] z=[%.2f, %.2f]' % (
        bbox.xmin, bbox.xmax, bbox.ymin, bbox.ymax,
        bbox.zmin, bbox.zmax))

    # Requested dimensions, converted from centimetres to millimetres.
    opening_width = 200.0
    opening_height = 100.0
    corner_radius = 10.0
    recess_depth = 30.0

    # Center the opening across the appliance width. A 50 mm lower margin is
    # close to the approximately 56 mm side margins resulting from a 200 mm
    # opening in the roughly 312 mm-wide enclosure.
    center_x = 0.5 * (bbox.xmin + bbox.xmax)
    bottom_margin = 50.0
    center_z = bbox.zmin + bottom_margin + 0.5 * opening_height

    # Positive Y is the rear. The principal rear-cover field is near y=415 mm;
    # cut 30 mm inward from that field and extend the tool outward far enough
    # to also trim ventilation ribs located in front of the cover.
    rear_surface_y = 415.0
    tool_inner_y = rear_surface_y - recess_depth
    tool_outer_y = bbox.ymax + 5.0
    tool_length = tool_outer_y - tool_inner_y

    rear_plane = cq.Plane(
        origin=(center_x, tool_outer_y, center_z),
        xDir=(1, 0, 0),
        normal=(0, -1, 0)
    )

    # Exact 200 x 100 mm rounded rectangle with 10 mm corner radii. It is
    # represented by overlapping simple prisms for robust subtraction from
    # imported STEP solids.
    cutter_parts = [
        cq.Workplane(rear_plane)
        .rect(opening_width - 2.0 * corner_radius, opening_height)
        .extrude(tool_length)
        .val(),
        cq.Workplane(rear_plane)
        .rect(opening_width, opening_height - 2.0 * corner_radius)
        .extrude(tool_length)
        .val()
    ]

    corner_x = 0.5 * opening_width - corner_radius
    corner_z = 0.5 * opening_height - corner_radius
    for dx in (-corner_x, corner_x):
        for dz in (-corner_z, corner_z):
            cutter_parts.append(
                cq.Workplane(rear_plane)
                .center(dx, dz)
                .circle(corner_radius)
                .extrude(tool_length)
                .val()
            )

    cutter_compound = cq.Compound.makeCompound(cutter_parts)
    cb = cutter_compound.BoundingBox()
    print('OPENING: width=%.2f height=%.2f radius=%.2f depth=%.2f mm' % (
        opening_width, opening_height, corner_radius, recess_depth))
    print('OPENING CENTER: x=%.2f z=%.2f' % (center_x, center_z))
    print('CUTTER BBOX: x=[%.2f, %.2f] y=[%.2f, %.2f] z=[%.2f, %.2f]' % (
        cb.xmin, cb.xmax, cb.ymin, cb.ymax, cb.zmin, cb.zmax))

    result_solids = []
    modified_count = 0
    split_count = 0
    total_removed = 0.0

    for index, solid in enumerate(source_shape.Solids()):
        sb = solid.BoundingBox()
        overlaps_cutter = not (
            sb.xmax < cb.xmin or sb.xmin > cb.xmax or
            sb.ymax < cb.ymin or sb.ymin > cb.ymax or
            sb.zmax < cb.zmin or sb.zmin > cb.zmax
        )

        if not overlaps_cutter:
            result_solids.append(solid)
            continue

        original_volume = solid.Volume()
        edited_shape = solid
        try:
            # Apply the opening to every solid that actually intersects the
            # rear cutting volume. Unlike the previous version, this includes
            # the ventilation ribs in front of the cover and therefore leaves
            # the opening envelope unobstructed.
            for cutter_part in cutter_parts:
                edited_shape = edited_shape.cut(cutter_part)

            edited_solids = list(edited_shape.Solids())
            remaining_volume = sum(piece.Volume() for piece in edited_solids)
            removed_volume = original_volume - remaining_volume

            if removed_volume > 1.0e-4:
                modified_count += 1
                total_removed += removed_volume
                if len(edited_solids) > 1:
                    split_count += 1
                print('CUT INTERSECTING SOLID %02d: removed %.3f mm^3, fragments=%d, bbox-y=[%.2f, %.2f]' % (
                    index, removed_volume, len(edited_solids), sb.ymin, sb.ymax))
                result_solids.extend(edited_solids)
            else:
                result_solids.append(solid)
        except Exception as exc:
            print('WARNING: cut failed for solid %02d: %s' % (index, str(exc)))
            result_solids.append(solid)

    result_shape = cq.Compound.makeCompound(result_solids)
    print('MODIFIED SOLIDS:', modified_count)
    print('SPLIT SOLIDS:', split_count)
    print('TOTAL REMOVED VOLUME: %.3f mm^3' % total_removed)
    print('RESULT SOLIDS:', len(result_shape.Solids()), 'FACES:', len(result_shape.Faces()))
    print('RESULT VALID:', result_shape.isValid())

    if modified_count == 0:
        print('WARNING: No solids intersected the rear opening cutter.')

    return cq.Workplane('XY').newObject([result_shape])
