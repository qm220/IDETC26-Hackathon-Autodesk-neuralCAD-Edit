            return False

        result = []
        for face in shape.Faces():
            for candidate in face.Edges():
                try:
                    same = edge.isSame(candidate)
                except Exception:
                    same = edge.hashCode() == candidate.hashCode()
                if same:
                    result.append(face)
                    break
        return result

        if len(faces) != 2:
            return False

        try:

            n2 = faces[1].normalAt(c)
