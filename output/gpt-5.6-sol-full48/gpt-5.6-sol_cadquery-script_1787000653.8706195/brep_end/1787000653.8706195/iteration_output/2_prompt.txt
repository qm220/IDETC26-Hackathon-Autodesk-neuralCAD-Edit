def my_cad_function(args):
    import os
    import math

    input_file = os.path.expanduser(args['input_file'])
    imported = cq.importers.importStep(input_file)
    shape = imported.val()
    solids = list(shape.Solids())

    if len(solids) != 2:
        raise RuntimeError('Expected the original model to contain exactly two solids, found %d' % len(solids))

    # The larger solid is the wheel body. The smaller solid is the separate
    # splined insert, which must remain unchanged.
    wheel = max(solids, key=lambda s: s.Volume())
    insert = min(solids, key=lambda s: s.Volume())
    original_wheel_volume = wheel.Volume()
    original_insert_volume = insert.Volume()

    wb = wheel.BoundingBox()
    axis_x = 0.5 * (wb.xmin + wb.xmax)
    axis_z = 0.5 * (wb.zmin + wb.zmax)
    front_y = wb.ymax

    # Locate the circular parent edge immediately below the existing small
    # front-center edge treatment. In this model it is the wheel-body cavity
    # circle at approximately Y=2.921 and R=15.826 mm. Selecting it
    # geometrically avoids relying on unstable imported STEP face numbers.
    candidates = []
    seen = set()
    for edge in wheel.Edges():
        key = edge.hashCode()
        if key in seen:
            continue
        seen.add(key)
        try:
            if edge.geomType() != 'CIRCLE':
                continue
            eb = edge.BoundingBox()
            ec = edge.Center()
            radius = 0.25 * ((eb.xmax - eb.xmin) + (eb.zmax - eb.zmin))
            axis_offset = math.hypot(ec.x - axis_x, ec.z - axis_z)
            axial_drop = front_y - ec.y
            if axis_offset < 0.05 and 14.0 < radius < 18.0 and 0.05 < axial_drop < 0.60:
                candidates.append((ec.y, radius, edge))
        except Exception:
            pass

    if not candidates:
        raise RuntimeError('Could not locate the front-center circular wheel-body edge')

    # The highest qualifying circle is the cylindrical parent edge beneath the
    # original rounded/small edge treatment, rather than an insert edge.
    candidates.sort(key=lambda item: item[0], reverse=True)
    edge_y, parent_radius, target_edge = candidates[0]

    chamfer_size = 1.0
    chamfer_bottom_y = front_y - chamfer_size
    chamfer_outer_radius = parent_radius + chamfer_size

    print('=== FRONT-CENTER CHAMFER EDIT ===')
    print('Wheel front Y: %.6f mm' % front_y)
    print('Located parent circle: Y=%.6f mm, R=%.6f mm' % (edge_y, parent_radius))
    print('Creating equal-distance chamfer: 1.000000 mm x 1.000000 mm')
    print('Chamfer endpoints in radial section: (R=%.6f,Y=%.6f) to (R=%.6f,Y=%.6f)' % (
        parent_radius, chamfer_bottom_y, chamfer_outer_radius, front_y))

    # Remove the former edge treatment and establish a true 1 x 1 mm, 45-degree
    # chamfer with a revolved triangular cut about the STEP Y-axis. The small
    # extension beyond the front plane improves boolean robustness without
    # altering the finished 1 mm intersection on the model surface.
    overshoot = 0.10
    cutter = (

        .moveTo(parent_radius, chamfer_bottom_y)
        .lineTo(parent_radius, front_y + overshoot)
        .lineTo(chamfer_outer_radius + overshoot, front_y + overshoot)
        .close()
        .revolve(360.0, (0.0, 0.0), (0.0, 1.0))
        .val()
    )
