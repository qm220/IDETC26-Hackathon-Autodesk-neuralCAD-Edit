    import math
    import cadquery as cq

    input_file = os.path.expanduser(args["input_file"])

    source = imported.val() if hasattr(imported, "val") else imported
    solids = list(source.Solids())

    if len(solids) < 3:
        raise ValueError("Expected two blade solids and one clamp/pivot solid in the input model")

    # Identify the two long blade bodies and retain the clamp/pivot body unchanged.
    ranked = sorted(solids, key=lambda s: max(s.BoundingBox().xlen, s.BoundingBox().zlen), reverse=True)
    blade_a = ranked[0]
    blade_b = ranked[1]
    hardware = ranked[2]

    stack_y = 6.35
    full_thickness = 12.70
    blade_width = 25.40
    web_thickness = 0.42
    layer_pitch = 0.47       # 0.42 mm material plus 0.05 mm running clearance
    central_half_length = 50.0
    web_half_length = 38.0
    pivot_clearance_radius = 6.55
    edge_radius = 3.175

    def axis_angle_xz(solid):
        """Find the undirected longitudinal angle of a blade from its longest linear edge."""
        best_len = -1.0
        best_angle = None
        for edge in solid.Edges():
            try:
                if edge.geomType() != "LINE":
                    continue
                vertices = edge.Vertices()
                if len(vertices) < 2:
                    continue
                p0 = vertices[0].Center()
                p1 = vertices[-1].Center()
                dx = p1.x - p0.x
                dz = p1.z - p0.z
                projected = math.hypot(dx, dz)
                if projected > best_len:
                    angle = math.degrees(math.atan2(dz, dx))
                    while angle >= 90.0:
                        angle -= 180.0
                    while angle < -90.0:
                        angle += 180.0
                    best_len = projected
                    best_angle = angle
            except Exception:
                pass
        if best_angle is None:
            bb = solid.BoundingBox()
            best_angle = 90.0 if bb.zlen > bb.xlen else 0.0
        return best_angle

    angle_a = axis_angle_xz(blade_a)
    angle_b = axis_angle_xz(blade_b)

    # Put the nearly vertical member second for predictable layer ordering.
    if abs(angle_a) > abs(angle_b):
        blade_a, blade_b = blade_b, blade_a
        angle_a, angle_b = angle_b, angle_a

    print("=== ROTOR EDIT ===")
    print("Detected existing blade angles: %.3f deg and %.3f deg" % (angle_a, angle_b))

    def rotate_local(shape, angle_deg):
        # A positive geometric X-Z angle requires a negative rotation about +Y.
        return shape.rotate((0, 0, 0), (0, 1, 0), -angle_deg)

    def central_cut_tool(angle_deg):

            60.0,
            cq.Vector(0, stack_y - 30.0, 0),
            cq.Vector(0, 1, 0),
        )

    def local_web(angle_deg, y_center):

        # radius compatible with the reduced thickness.
        wp = (
            cq.Workplane("XY")
            .box(2.0 * web_half_length, web_thickness, blade_width)
            .translate((0, y_center, 0))
        )
        try:
            web = wp.edges("|X").fillet(0.18).val()
        except Exception:
            web = wp.val()
        web = rotate_local(web, angle_deg).cut(pin_cut)
        return web

            .center(y1 - y0, 0)
            .rect(full_thickness, blade_width)
            .loft(combine=True)
            .val()
        )
        return rotate_local(loft, angle_deg).cut(pin_cut)

    def replacement_center(angle_deg, layer_center):
        parts = [local_web(angle_deg, layer_center)]
        parts.append(
            local_transition(
                web_half_length,
                central_half_length + 0.5,
                layer_center,
                stack_y,

        parts.append(
            local_transition(
                -web_half_length,
                -(central_half_length + 0.5),

                stack_y,
                angle_deg,

    edited_b = blade_b.cut(central_cut_tool(angle_b))

    angle_new = angle_b - 120.0

    )
    total_length = max(360.0, min(381.0, source_span))
    half_length = total_length / 2.0

    def new_outer_segment(x_min, x_max):
        length = x_max - x_min

            body = body_wp.edges("|X").fillet(edge_radius).val()
        except Exception as exc:

            body = body_wp.val()

        # Reproduce the source blade's longitudinal hollow bore.
        bore = cq.Solid.makeCylinder(
            3.175,
            length + 2.0,
            cq.Vector(x_min - 1.0, stack_y, 0),
            cq.Vector(1, 0, 0),
        )
        return body.cut(bore)

    negative_outer = new_outer_segment(-half_length, -(central_half_length - 0.25))
    positive_outer = new_outer_segment(central_half_length - 0.25, half_length)
    result_shapes.append(rotate_local(negative_outer, angle_new))
    result_shapes.append(rotate_local(positive_outer, angle_new))
    result_shapes.extend(replacement_center(angle_new, layer_new))

    result = cq.Compound.makeCompound(result_shapes)
    print("Result valid:", result.isValid())
    print("Result solids:", len(result.Solids()))
    print("Central web thickness: %.3f mm" % web_thickness)
    print("Central layer centers: %.3f, %.3f, %.3f mm" % (layer_a, layer_new, layer_b))
    print("New blade long-edge radius: %.3f mm" % edge_radius)

    return cq.Workplane("XY").newObject([result])
