    import cadquery as cq
    from OCP.BRepAlgoAPI import BRepAlgoAPI_Defeaturing

    input_file = os.path.expanduser(args["input_file"])
    imported = cq.importers.importStep(input_file)
    imported_shape = imported.val()

    solids = imported_shape.Solids()
    if len(solids) != 1:
        raise RuntimeError("Expected one solid in the input STEP, found %d" % len(solids))
    original = solids[0]

    # Identify the eight faces comprising the continuous R10 outer bullnose.
    # R50 and R63 are planform corner radii and must remain unchanged.
    target_faces = []
    for face in original.Faces():
        for edge in face.Edges():
            if edge.geomType() != "CIRCLE":
                continue
            try:
                if abs(edge.radius() - 10.0) < 1.0e-4:
                    target_faces.append(face)
                    break
            except Exception:
                pass

    print("R10 bullnose faces selected:", len(target_faces))
    if len(target_faces) != 8:
        raise RuntimeError(
            "Expected the 8 faces of the continuous R10 bullnose, found %d"
            % len(target_faces)
        )

    # Store the rear retaining-land plane so it can be recovered after healing.
    annular_faces = [
        face for face in original.Faces()
        if face.geomType() == "PLANE" and len(face.Wires()) >= 2
    ]
    if len(annular_faces) < 2:
        raise RuntimeError("Could not identify both planar annular lands")

    rear_land = min(annular_faces, key=lambda face: face.Area())
    rear_normal = rear_land.normalAt()
    rear_center = rear_land.Center()
    rear_plane_offset = (
        rear_center.x * rear_normal.x
        + rear_center.y * rear_normal.y
        + rear_center.z * rear_normal.z
    )

    # Remove the complete R10 feature and extend/heal its supporting faces.
    defeature = BRepAlgoAPI_Defeaturing()

        raise RuntimeError("OCCT failed to remove and heal the R10 bullnose")

            "R10 removal did not produce one solid; found %d" % len(healed_solids)

        raise RuntimeError("The solid produced by R10 removal is invalid")

    ))

    # Recover the healed rear annular land using its original plane equation.
    rear_candidates = []
    for face in healed.Faces():

            continue

        normal = face.normalAt()
        alignment = abs(
            normal.x * rear_normal.x
            + normal.y * rear_normal.y
            + normal.z * rear_normal.z
        )

        plane_error = abs(
            center.x * rear_normal.x
            + center.y * rear_normal.y
            + center.z * rear_normal.z
            - rear_plane_offset
        )
        if alignment > 0.999999 and plane_error < 1.0e-3:
            rear_candidates.append(face)

    if not rear_candidates:
        raise RuntimeError("Could not recover the healed rear retaining land")

    healed_rear = max(rear_candidates, key=lambda face: face.Area())
    rear_wires = healed_rear.Wires()
    if len(rear_wires) < 2:
        raise RuntimeError("Recovered rear land is not annular")

    # The longest wire is the outer all-around edge where R10 was removed.
    outer_wire = max(rear_wires, key=lambda wire: wire.Length())

    result = healed.makeFillet(2.0, perimeter_edges).clean()

            "R2 replacement did not produce one solid; found %d"

            if abs(radius - 2.0) < 1.0e-4:
                r2_edges += 1
        except Exception:
            pass

    print("FINAL VALID", result.isValid())

    print("FINAL COUNTS solids=%d faces=%d edges=%d" % (
