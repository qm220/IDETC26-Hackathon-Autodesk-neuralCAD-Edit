        f"Main solid BBox: xmin={bb.xmin:.3f} xmax={bb.xmax:.3f} ymin={bb.ymin:.3f} ymax={bb.ymax:.3f} "

            continue

        fb = f.BoundingBox()

        # Axis ~Y => xlen ~ zlen

        radial_center = math.hypot(fc.x - xmid, fc.z - zmid)
