        nsol = len(list(shp.Solids()))
    except Exception:
        nsol = None

    bbox = shp.BoundingBox()
    print("Loaded STEP:", step_path)
    print("Solids in working shape:", nsol)
    print(f"BBOX xmin/xmax: {bbox.xmin:.3f}, {bbox.xmax:.3f}")
    print(f"BBOX ymin/ymax: {bbox.ymin:.3f}, {bbox.ymax:.3f}")
    print(f"BBOX zmin/zmax: {bbox.zmin:.3f}, {bbox.zmax:.3f}")

    faces = list(shp.Faces())

    def _safe_normal(face):
        try:
            n = face.normalAt()
            return cq.Vector(n.x, n.y, n.z)
        except Exception:
            try:
                ad = face._geomAdaptor()
                if face.geomType() == "PLANE":
                    pln = ad.Plane()
                    d = pln.Axis().Direction()
                    return cq.Vector(d.X(), d.Y(), d.Z())
            except Exception:
                pass
        return cq.Vector(0, 0, 0)

    # --- detect arm side planes (Y) and arm thickness planes (Z) ---
    plane_faces_y = []
    plane_faces_z = []
    for f in faces:
        if f.geomType() != "PLANE":
            continue
        n = _safe_normal(f)
        c = f.Center()
        a = f.Area()
        if abs(n.y) > 0.95 and abs(n.x) < 0.25 and abs(n.z) < 0.25:
            plane_faces_y.append((a, c.y))
        if abs(n.z) > 0.95 and abs(n.x) < 0.25 and abs(n.y) < 0.25:
            plane_faces_z.append((a, c.z))

    plane_faces_y.sort(reverse=True, key=lambda t: t[0])
    plane_faces_z.sort(reverse=True, key=lambda t: t[0])

    if len(plane_faces_y) >= 2:
        y_lower, y_upper = sorted([plane_faces_y[0][1], plane_faces_y[1][1]])
    else:
        y_lower, y_upper = bbox.ymin, bbox.ymax

    if len(plane_faces_z) >= 2:
        z_bottom, z_top = sorted([plane_faces_z[0][1], plane_faces_z[1][1]])
    else:
        z_bottom, z_top = bbox.zmin, bbox.zmax

    z_mid = 0.5 * (z_bottom + z_top)
    arm_width_y = (y_upper - y_lower)
    arm_thk_z = (z_top - z_bottom)

    print(f"Detected arm Y side planes: y_lower={y_lower:.3f}, y_upper={y_upper:.3f}")
    print(f"Detected arm Z planes: z_bottom={z_bottom:.3f}, z_top={z_top:.3f}")

    # --- detect boss + bore cylinders (axis ~Y) ---
    cyl_info = []
    for f in faces:
        if f.geomType() != "CYLINDER":
            continue
        try:
            ad = f._geomAdaptor()
            cyl = ad.Cylinder()
            r = float(cyl.Radius())
            d = cyl.Axis().Direction()
            axis = cq.Vector(d.X(), d.Y(), d.Z())
            c = f.Center()
            bb = f.BoundingBox()
            cyl_info.append((r, axis, c, bb))
        except Exception:
            continue

    def _axis_is_y(ax: cq.Vector):
        return abs(ax.y) > 0.92 and abs(ax.x) < 0.35 and abs(ax.z) < 0.35

    boss_cands = [ci for ci in cyl_info if _axis_is_y(ci[1]) and 6.3 <= ci[0] <= 7.7]
    bore_cands = [ci for ci in cyl_info if _axis_is_y(ci[1]) and 4.3 <= ci[0] <= 5.7]

    if not boss_cands:
        raise ValueError("Could not find boss outer cylinder (r~7, axis~Y).")
    if not bore_cands:
        raise ValueError("Could not find bore cylinder (r~5, axis~Y).")

    # choose boss with highest +Y extent (the socket mouth side)
    boss_cands.sort(key=lambda t: t[3].ymax, reverse=True)
    boss_r, boss_axis, boss_center, boss_bb = boss_cands[0]

    # choose bore closest in XZ to boss center
    def _xz_dist(ci):
        c = ci[2]
        return (c.x - boss_center.x) ** 2 + (c.z - boss_center.z) ** 2

    bore_cands.sort(key=_xz_dist)
    bore_r, bore_axis, bore_center, bore_bb = bore_cands[0]

    x0 = float(boss_center.x)
    z0 = float(boss_center.z)

    y_top = float(boss_bb.ymax)        # boss mouth plane (~34)
    bore_ymin = float(bore_bb.ymin)    # bore bottom (~25)

    print(f"Boss detected: r_outer={boss_r:.3f}, center=({x0:.3f},{boss_center.y:.3f},{z0:.3f}), y_start={y_cyl_start:.3f}, y_top={y_top:.3f}")
    print(f"Bore detected: r_inner={bore_r:.3f}, bore_ymin={bore_ymin:.3f}")

    # --- compute 4 boss X locations (keep original at x0, add 3 to +X) ---
    # keep clearance from right end
    right_clear = float(bbox.xmax - (boss_r + 3.0))
    avail = right_clear - x0
    if avail <= (2 * boss_r + 6.0):
        raise ValueError("Not enough room to place 3 additional bosses to +X while keeping clearance.")

    dx = avail / 3.0
    x_positions = [x0 + i * dx for i in range(4)]
    # clamp last to right_clear
    x_positions[-1] = min(x_positions[-1], right_clear)

    # New bosses: exclude original (within tolerance)
    tol = 0.25
    new_x_positions = [xp for xp in x_positions if abs(xp - x0) > tol]

    print("Boss pattern x positions (target 4 total):", [round(v, 3) for v in x_positions])
    print("New boss x positions to add:", [round(v, 3) for v in new_x_positions])

    # --- build new bosses (add) and bores (cut) ---
    add_all = None
    cut_all = None

    # embed slightly into arm for reliable boolean fuse
    embed = max(0.8, 0.06 * arm_width_y)

    boss_height = y_top - boss_base_y

    for xp in new_x_positions:
        outer = (
            cq.Workplane("XZ")
            .workplane(offset=boss_base_y)
            .center(xp, z0)
            .circle(boss_r)
            .extrude(boss_height)
        )
        add_all = outer if add_all is None else add_all.union(outer)

        # blind bore (open at y_top, bottom at bore_ymin)
        bore_len = max(0.2, y_top - bore_ymin)
        bore = (
            cq.Workplane("XZ")
            .workplane(offset=bore_ymin)
            .center(xp, z0)
            .circle(bore_r)
            .extrude(bore_len)
        )
        cut_all = bore if cut_all is None else cut_all.union(bore)

    # --- stability ribs/legs on the sides ---
    # Create compact leading/trailing gusset fins on both arm side faces.
    # +Y ribs are intended to intersect the boss (boss is on +Y side in the source model).
    dx_nom = x_positions[1] - x_positions[0] if len(x_positions) >= 2 else 25.0
    rib_run = min(8.0, max(4.0, 0.25 * dx_nom))  # along X

    # ribs protrusion depth in Y from each side face
    rib_depth_plus = min(6.0, max(3.5, (y_cyl_start - y_upper) + 4.0))
    rib_depth_minus = 4.0

    # overlap slightly into arm to ensure fusion
    rib_overlap = 0.35
    y_plus_start = y_upper - rib_overlap
    y_minus_start = y_lower + rib_overlap

    ribs_all = None

    def _clamp(xv, lo, hi):
        return max(lo, min(hi, xv))

    # Make ribs for each boss location (including original)
    for xp in x_positions:
        for sign in (-1.0, 1.0):  # leading/trailing
            base_x = xp + sign * boss_r
            tip_x = xp + sign * (boss_r + rib_run)

            # keep ribs within arm length envelope
            tip_x = _clamp(tip_x, bbox.xmin + 1.0, bbox.xmax - 1.0)

                .workplane(offset=y_plus_start)
                .polyline(tri)
                .close()
                .extrude(rib_depth_plus + rib_overlap)

            rib_minus = (
                cq.Workplane("XZ")
                .workplane(offset=y_minus_start)
                .polyline(tri)
                .close()
                .extrude(-(rib_depth_minus + rib_overlap))
            )

            ribs_all = rib_plus.union(rib_minus) if ribs_all is None else ribs_all.union(rib_plus).union(rib_minus)

    if ribs_all is not None:
        add_all = ribs_all if add_all is None else add_all.union(ribs_all)

    # --- apply booleans to the full imported shape ---
    result = cq.Workplane(obj=shp)
    if add_all is not None:
        result = result.union(add_all, clean=True)
    if cut_all is not None:
        result = result.cut(cut_all)

    # Try to fuse anything that is merely touching/overlapping
    try:
        result = result.combineSolids()
    except Exception as e:
        print("combineSolids after booleans failed (non-fatal):", e)

    # Best-effort edge softening for manufacturability
    try:
        # small fillet on sharp rib edges; keep conservative to reduce failures
        result = result.edges("|Y").fillet(0.35)
    except Exception as e:
        print("Fillet failed (non-fatal):", e)

    try:
        result = result.clean()
    except Exception:
        pass

    return result
