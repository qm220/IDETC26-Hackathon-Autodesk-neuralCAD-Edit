def my_cad_function(args):
    import cadquery as cq
    import os

    # ---- Load model ----
    input_file = os.path.expanduser(args.get("input_file", ""))
    shape_wp = cq.importers.importStep(input_file)
    shp = shape_wp.val() if hasattr(shape_wp, "val") else shape_wp

    bb = shp.BoundingBox()
    print(f"Loaded STEP: {input_file}")
    print(f"Valid: {shp.isValid()}")
    print(f"Faces: {len(shp.Faces())}, Solids: {len(shp.Solids())}")
    print(f"BBox xmin/xmax: {bb.xmin:.3f}/{bb.xmax:.3f}")
    print(f"BBox ymin/ymax: {bb.ymin:.3f}/{bb.ymax:.3f}")
    print(f"BBox zmin/zmax: {bb.zmin:.3f}/{bb.zmax:.3f}")
    print(f"BBox center: ({bb.center.x:.3f}, {bb.center.y:.3f}, {bb.center.z:.3f})")

    # ---- Find top tank reference: largest planar face among >Y faces ----
    wp0 = cq.Workplane(obj=shp)
    top_faces = wp0.faces(">Y").vals()
    planar = [f for f in top_faces if hasattr(f, "geomType") and f.geomType() == "PLANE"]

    if not planar:
        # fallback: just use bbox.ymax
        y_top_plane = bb.ymax
        print("WARNING: No planar >Y faces found; using bbox.ymax as top plane.")
    else:
        top_planar = max(planar, key=lambda f: f.Area())
        c = top_planar.Center()
        y_top_plane = c.y
        print(f"Top planar face chosen: area={top_planar.Area():.3f}, center=({c.x:.3f},{c.y:.3f},{c.z:.3f})")

    # Neck placement: near longitudinal mid-span (z~0) and centered in x.
    # Use bbox center x,z as robust defaults.
    cx = bb.center.x
    cz = bb.center.z
    print(f"Neck center target (cx,cz)=({cx:.3f},{cz:.3f}) at y={y_top_plane:.3f}")

    # ---- Parameters (mm) ----
    neck_od = 42.0
    neck_id = 32.0
    neck_h = 28.0
    wall = (neck_od - neck_id) / 2.0

    cut_depth_into_tank = 20.0  # cut down (-Y) from top plane

    flange_h = 3.0
    flange_od = 52.0

    # Simple bayonet-lug approximation
    lug_len = 14.0   # along X
    lug_depth = 8.0  # along Z
    lug_thk = 2.5    # along +Y
    lug_y = y_top_plane + neck_h - lug_thk  # near neck top

    # Cap (simplified)
    cap_od = 62.0
    cap_h = 25.0
    cap_insert_depth = 14.0
    cap_clearance = 0.8

    # ---- Build features on a global XZ workplane located at the top tank plane ----
    top_wpl = cq.Workplane("XZ", origin=(0, y_top_plane, 0)).center(cx, cz)

    # 1) Create pouring opening into tank
    cut_tool = top_wpl.circle(neck_id / 2.0).extrude(-cut_depth_into_tank)
    modified = shape_wp.cut(cut_tool)

    # 2) Create neck tube
    neck = top_wpl.circle(neck_od / 2.0).circle(neck_id / 2.0).extrude(neck_h)

    # 3) Add a small top flange/seat
    flange_wpl = cq.Workplane("XZ", origin=(0, y_top_plane + neck_h - flange_h, 0)).center(cx, cz)
    flange = flange_wpl.circle(flange_od / 2.0).circle(neck_od / 2.0).extrude(flange_h)

    # 4) Add two simple bayonet lugs (visual/functional placeholder)
    lugs_wpl = cq.Workplane("XZ", origin=(0, lug_y, 0)).center(cx, cz)
    lug_r = neck_od / 2.0 + lug_depth / 2.0
    lugs = (
        lugs_wpl
        .pushPoints([(0, lug_r), (0, -lug_r)])
        .rect(lug_len, lug_depth)
        .extrude(lug_thk)
    )

    # Union neck features onto radiator
    modified = modified.union(neck).union(flange).union(lugs)

    # ---- Create cap as separate body (simplified) ----
    cap_y0 = y_top_plane + neck_h - cap_insert_depth  # cap bottom start
    cap_wpl = cq.Workplane("XZ", origin=(0, cap_y0, 0)).center(cx, cz)
    cap_solid = cap_wpl.circle(cap_od / 2.0).extrude(cap_h)

    # Hollow inside to fit over neck
    inner_id = neck_od + 2.0 * cap_clearance
    inner_cut = cap_wpl.circle(inner_id / 2.0).extrude(cap_insert_depth + 6.0)
    cap_solid = cap_solid.cut(inner_cut)

    # Add a shallow top recess for visual detail
    recess = cq.Workplane("XZ", origin=(0, cap_y0 + cap_h - 3.0, 0)).center(cx, cz).circle((cap_od - 10.0) / 2.0).extrude(3.0)
    cap_solid = cap_solid.cut(recess)

    # ---- Return as assembly so cap is separate ----
    asm = cq.Assembly(name="radiator_with_filler")
    asm.add(modified, name="radiator", color=cq.Color(0.2, 0.2, 0.2))
    asm.add(cap_solid, name="cap", color=cq.Color(0.1, 0.1, 0.1))

    print("Filler neck + opening + simplified bayonet lugs + separate cap created.")
    return asm
