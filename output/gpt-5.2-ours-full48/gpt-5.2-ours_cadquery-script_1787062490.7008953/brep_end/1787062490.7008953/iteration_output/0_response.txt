def my_cad_function(args):
    import cadquery as cq
    import os

    input_file = os.path.expanduser(args.get("input_file", ""))
    if not input_file or (not os.path.exists(input_file)):
        raise ValueError(f"Input STEP file not found: {input_file}")

    # Load model
    imp = cq.importers.importStep(input_file)
    shape = imp.val() if hasattr(imp, "val") else imp

    # Extract solids
    solids = cq.Workplane(obj=shape).solids().vals()
    print(f"Loaded STEP: {input_file}")
    print(f"Total solids found: {len(solids)}")

    # Identify the 'Nozzle Volcano' (S9) by location ~ (x=40, y=-60) and height ~ 21
    target_x, target_y, target_zmax = 40.0, -60.0, 21.0
    best_i = None
    best_score = None

    for i, s in enumerate(solids):
        bb = s.BoundingBox()
        c = bb.center
        # Weighted score: strong XY proximity, mild Z proximity
        dxy2 = (c.x - target_x) ** 2 + (c.y - target_y) ** 2
        score = dxy2 + 0.5 * (bb.zmax - target_zmax) ** 2

        print(
            f"Solid[{i}] bb: x=({bb.xmin:.3f},{bb.xmax:.3f}) y=({bb.ymin:.3f},{bb.ymax:.3f}) z=({bb.zmin:.3f},{bb.zmax:.3f}) "
            f"center=({c.x:.3f},{c.y:.3f},{c.z:.3f}) score={score:.3f}"
        )

        # Additional coarse filter to avoid selecting main blocks
        if bb.ymax < -40 and bb.xmin > 10 and bb.xmax < 80 and bb.zmax < 40:
            if best_score is None or score < best_score:
                best_score = score
                best_i = i

    if best_i is None:
        # fallback: pick absolute best score without coarse filter
        for i, s in enumerate(solids):
            bb = s.BoundingBox()
            c = bb.center
            dxy2 = (c.x - target_x) ** 2 + (c.y - target_y) ** 2
            score = dxy2 + 0.5 * (bb.zmax - target_zmax) ** 2
            if best_score is None or score < best_score:
                best_score = score
                best_i = i

    nozzle = solids[best_i]
    bb = nozzle.BoundingBox()
    print(f"\nSelected nozzle solid index: {best_i}")
    print(
        f"Selected nozzle bb: x=({bb.xmin:.3f},{bb.xmax:.3f}) y=({bb.ymin:.3f},{bb.ymax:.3f}) z=({bb.zmin:.3f},{bb.zmax:.3f})"
    )

    # Task: Reduce overall height by 1mm, taken from threaded area.
    # Robust implementation: truncate ONLY this nozzle by removing a 1mm slice from the TOP.
    height_reduction = 1.0
    new_zmax = bb.zmax - height_reduction

    print(f"Truncating selected nozzle: zmax {bb.zmax:.3f} -> {new_zmax:.3f} (reduction {height_reduction:.3f} mm)")

    # Create a large 'keep' box that includes everything below new_zmax, then intersect.
    keep_x = max(200.0, (bb.xlen + 50.0))
    keep_y = max(200.0, (bb.ylen + 50.0))
    keep_z = 200.0
    # Box centered so that its TOP face is exactly at new_zmax
    keep_box = (
        cq.Workplane("XY")
        .box(keep_x, keep_y, keep_z, centered=(True, True, True))
        .translate((0, 0, new_zmax - keep_z / 2.0))
        .val()
    )

    nozzle_short = nozzle.intersect(keep_box)
    bb2 = nozzle_short.BoundingBox()
    print(
        f"Result nozzle bb: x=({bb2.xmin:.3f},{bb2.xmax:.3f}) y=({bb2.ymin:.3f},{bb2.ymax:.3f}) z=({bb2.zmin:.3f},{bb2.zmax:.3f})"
    )

    # Recombine into a single compound with the modified nozzle
    new_solids = []
    for i, s in enumerate(solids):
        new_solids.append(nozzle_short if i == best_i else s)

    result = cq.Compound.makeCompound(new_solids)
    return result
