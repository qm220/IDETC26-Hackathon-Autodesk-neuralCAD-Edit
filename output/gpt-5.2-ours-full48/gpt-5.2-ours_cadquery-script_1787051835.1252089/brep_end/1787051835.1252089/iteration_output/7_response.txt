    from OCP.BRepFilletAPI import BRepFilletAPI_MakeFillet

    head_x_limit = xmin + min(102.0, 0.38 * xlen)

    tolY = max(0.40, 0.01 * ylen)

            return e.geomType() == "LINE"
        except Exception:
            return False

        try:
            return e.Center()

        eb = e.BoundingBox()

            near(eb.ymin, ymin, tol) or near(eb.ymax, ymax, tol) or
