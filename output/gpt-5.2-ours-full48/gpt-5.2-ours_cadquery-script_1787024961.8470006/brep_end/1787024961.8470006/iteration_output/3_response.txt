def my_cad_function(args):

    import os, math, statistics

    if "input_file" not in args:
        raise ValueError("args['input_file'] is required")

    input_path = os.path.expanduser(args["input_file"])
    imported = cq.importers.importStep(input_path)

    # Normalize import to a single Shape
    if isinstance(imported, cq.Workplane):
        root = imported.val()
    elif hasattr(cq, "Assembly") and isinstance(imported, cq.Assembly):
        root = imported.toCompound()
    else:
        root = imported

    wp = cq.Workplane(obj=root)

    # --- pick main solid (for inside/outside tests) ---
    solids = []
    try:
        solids = list(root.Solids())
    except Exception:
        try:
            solids = wp.solids().vals()
        except Exception:
            solids = []

    if solids:
        main_solid = max(solids, key=lambda s: s.Volume())
    else:
        # fall back: assume root is a solid-like shape
        main_solid = root

    # --- units heuristic from bbox (kept simple) ---
    try:
        bb = root.BoundingBox()
        max_dim = max(bb.xlen, bb.ylen, bb.zlen)
        # This dataset often comes in inches; bbox here is clearly small.
        units = "inch" if max_dim < 100 else "mm"
    except Exception:
        units = "inch"

    chamfer_mm = 0.2
    chamfer_size = chamfer_mm if units == "mm" else chamfer_mm / 25.4

    def edge_is_full_circle(e, r, rel_tol=0.03):
        """Heuristic: full circle edge has length ~ 2*pi*r."""
        try:
            L = float(e.Length())
            return abs(L - 2.0 * math.pi * r) <= (2.0 * math.pi * r) * rel_tol
        except Exception:
            return False

    def try_is_inside(solid, p):
        """Return True if point is inside solid, False if outside/unknown."""
        try:
            # CQ/OCCT typically supports isInside(point, tol)
            return bool(solid.isInside(p, 1e-6))
        except Exception:
            try:
                return bool(solid.isInside(p))
            except Exception:
                return False

    def dedup_edges(edges):
        seen = set()
        out = []
        for e in edges:
            try:
                k = e.hashCode(2147483647)
            except Exception:
                k = id(e)
            if k in seen:
                continue
            seen.add(k)
            out.append(e)
        return out

    # --- find INTERNAL through-hole cylinders by analyzing cylindrical faces ---
    cyl_faces = []
    for f in root.Faces():
        try:
            if f.geomType() != "CYLINDER":
                continue

            continue

        # collect full-circle boundary edges on this cylindrical face
        circ_edges = []
        circ_r = []
        for e in f.Edges():

                if e.geomType() != "CIRCLE":
                    continue
                r = float(e.radius())

            if edge_is_full_circle(e, r):

        if len(circ_edges) != 2:

        # radius consistency
        r_med = statistics.median(circ_r)
        if max(abs(rr - r_med) for rr in circ_r) > max(1e-4, 0.002 * r_med):
            continue

        # cylinder "length" estimated by distance between circle centers
        try:
            c0 = circ_edges[0].Center()
            c1 = circ_edges[1].Center()
            length = (c1.sub(c0)).Length
        except Exception:
            length = None

        # internal-vs-external test: axis point (midpoint between circle centers)
        try:
            mid = cq.Vector((c0.x + c1.x) * 0.5, (c0.y + c1.y) * 0.5, (c0.z + c1.z) * 0.5)

    # Pin group: prefer count==2 and radius larger than mount
    candidates_2 = [(k, groups[k]) for k in groups if len(groups[k]) == 2]
    if candidates_2:
        if mount_r is not None:
            bigger = [kg for kg in candidates_2 if statistics.median([c["r"] for c in kg[1]]) > mount_r * 1.15]
            if bigger:
                pin_group = min(bigger, key=lambda kg: statistics.median([c["r"] for c in kg[1]]))[1]
            else:
                pin_group = min(candidates_2, key=lambda kg: statistics.median([c["r"] for c in kg[1]]))[1]
        else:

    # Fallback: if grouping heuristics fail, just chamfer all detected internal through-hole cylinder ends
    selected = []

        selected.extend(pin_group)
    if not selected:
        selected = cyl_faces

    # Collect the end-circle edges for chamfer (both ends of each hole cylinder)
    target_edges = []
    for c in selected:
        target_edges.extend(c["edges"])

    target_edges = dedup_edges(target_edges)

    radii = []

        try:
            radii.append(float(e.radius()))

            pass

        result = wp.newObject(target_edges).chamfer(chamfer_size)
        print("[DBG] Chamfer applied to hole edges.")
        return result
