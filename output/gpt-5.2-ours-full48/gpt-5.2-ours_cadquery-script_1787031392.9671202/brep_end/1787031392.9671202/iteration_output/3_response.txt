    import os, math

    input_file = os.path.expanduser(args.get("input_file", ""))

        print(f"ERROR: input_file not found: {input_file}")

    # --- Load STEP ---
    try:
        wp_import = cq.importers.importStep(input_file)
    except Exception as e:
        print(f"ERROR importing STEP: {e}")
        return None

    shape = wp_import.val() if hasattr(wp_import, "val") else wp_import
    print(f"Loaded shape. Valid={shape.isValid()}")

    bbox = shape.BoundingBox()
    print(
        "Global BBox: "
        f"xmin={bbox.xmin:.3f} xmax={bbox.xmax:.3f} "
        f"ymin={bbox.ymin:.3f} ymax={bbox.ymax:.3f} "
        f"zmin={bbox.zmin:.3f} zmax={bbox.zmax:.3f}"
    )

    # --- Helpers ---
    def _is_planar(face):
        try:
            return face.geomType() == "PLANE"
        except Exception:
            return False

    def _area(face):
        try:
            return face.Area()
        except Exception:
            return 0.0

    def _safe_normal(face):
        try:
            return face.normalAt().normalized()
        except Exception:
            return None

    def _build_plane_on_face(face):
        c = face.Center()
        n = _safe_normal(face)
        if n is None:
            n = cq.Vector(0, 1, 0)
        xDir = cq.Vector(1, 0, 0)
        if abs(n.dot(xDir)) > 0.98:
            xDir = cq.Vector(0, 0, 1)
        return cq.Plane(origin=c, xDir=xDir, normal=n)

    def _circle_keepouts_on_plane_y(y0, face_bb, slot_w, slot_len, y_tol=0.35):
        """Find circle edges that lie (approximately) on a given Y plane; return list of (cx, cz, r_keepout)."""
        keepouts = []
        # bounds check in XZ to avoid unrelated circular edges elsewhere
        xpad = 5.0
        zpad = 5.0
        xmin, xmax = face_bb.xmin - xpad, face_bb.xmax + xpad
        zmin, zmax = face_bb.zmin - zpad, face_bb.zmax + zpad

        # reasonable radius bounds based on face extents
        xSpan = face_bb.xmax - face_bb.xmin
        zSpan = face_bb.zmax - face_bb.zmin
        r_min = 4.0
        r_max = 0.95 * min(xSpan, zSpan) if min(xSpan, zSpan) > 1e-6 else 200.0

        pad = (slot_w / 2.0) + 6.0  # keep slots away from bosses/holes

        for e in shape.Edges():
            try:
                if e.geomType() != "CIRCLE":
                    continue
            except Exception:
                continue

            try:
                ebb = e.BoundingBox()
                if (ebb.ymax - ebb.ymin) > (2.0 * y_tol):
                    continue
                # if the circle edge is on (or very near) the requested y-plane
                if abs(0.5 * (ebb.ymax + ebb.ymin) - y0) > y_tol:
                    continue

                circ = e._geomAdaptor().Circle()
                r = float(circ.Radius())
                center = circ.Location()
                cx, cy, cz = float(center.X()), float(center.Y()), float(center.Z())
            except Exception:
                continue

            if abs(cy - y0) > (2.0 * y_tol):
                continue
            if r < r_min or r > r_max:
                continue
            if not (xmin <= cx <= xmax and zmin <= cz <= zmax):
                continue

            rk = r + pad

            # Deduplicate near-identical circles
            merged = False
            for i, (kx, kz, kr) in enumerate(keepouts):
                if math.hypot(cx - kx, cz - kz) < 1.0 and abs(rk - kr) < 1.0:
                    merged = True
                    break
            if not merged:

        # Compute safe extents so slots don't clip boundaries
        x0 = fbb.xmin + edge_margin + (slot_w / 2.0)
        x1 = fbb.xmax - edge_margin - (slot_w / 2.0)

        z1 = fbb.zmax - edge_margin - (slot_len / 2.0)

        if z1 <= z0:
            print(f"Face {tag}: no Z room for slots after margins")
            return plane, []

        # X: if not enough width, use a single centered column
        if x1 <= x0:
            xs = [0.5 * (fbb.xmin + fbb.xmax)]
        else:
            xs = []
            x = x0
            while x <= x1 + 1e-6:
                xs.append(x)
                x += pitch_x
            if not xs:
                xs = [0.5 * (x0 + x1)]

        zs = []

        except TypeError:
            w = w.slot2D(slot_len, slot_w)

    return base
