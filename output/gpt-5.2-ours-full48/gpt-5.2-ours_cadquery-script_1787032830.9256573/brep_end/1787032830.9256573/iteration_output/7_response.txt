    front_margin = 6.0          # keep away from nose end
    wheel_clear = 3.0           # keep-out distance from wheel along length axis (Z)

    draft_shrink = 1.0          # shrink top footprint relative to bottom (per side)
    top_edge_fillet = 0.6       # comfort fillet on top edges (best-effort)

    def clamp(v, lo, hi):
        return max(lo, min(hi, v))

    # Assume: X=width, Y=up, Z=length
    cx = 0.5 * (body_bb.xmin + body_bb.xmax)

        if nose_is_zmax:

            )

        except Exception as e:
            print(f"Warning: loft failed at xc={xc_local:.2f} (will use extrude fallback): {e}")

    # Preserve wheel as separate solid if present
    if wheel is not None:
        result = cq.Compound.makeCompound([body_mod, wheel])
    else:
        result = body_mod

    return result
