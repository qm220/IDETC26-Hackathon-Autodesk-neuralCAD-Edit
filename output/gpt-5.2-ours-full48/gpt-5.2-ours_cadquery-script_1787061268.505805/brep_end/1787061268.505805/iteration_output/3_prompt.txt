    def wire_circle_data(w):

        return face, circles

    def cluster_1d(vals, tol=0.8):
        if not vals:

            if abs(v - cmean) <= tol:
                clusters[-1].append(v)
            else:
                clusters.append([v])
        return [sum(c) / len(c) for c in clusters]
