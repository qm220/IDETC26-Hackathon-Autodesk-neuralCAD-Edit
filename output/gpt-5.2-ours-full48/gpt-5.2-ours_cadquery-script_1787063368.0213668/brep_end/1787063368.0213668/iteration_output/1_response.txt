        outer = face.outerWire()
        inners = []
        for w in face.Wires():
            try:
                if not w.isSame(outer):
                    inners.append(w)
            except Exception:
                if w.hashCode() != outer.hashCode():
                    inners.append(w)
        return inners

                else:
                    out.append(cq.Vector(p.X(), p.Y(), p.Z()))
            return out
        except Exception:

            except Exception:

        if L < 1e-9:
            return None
        xdir = cq.Vector(v.x / L, v.y / L, 0)

                y = vv.dot(ydir)
                max_abs_y = max(max_abs_y, abs(y))
