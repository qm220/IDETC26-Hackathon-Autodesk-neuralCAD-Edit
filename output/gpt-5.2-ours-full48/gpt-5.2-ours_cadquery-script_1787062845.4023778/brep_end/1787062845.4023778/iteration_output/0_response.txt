    import os

    if "input_file" not in args:
        raise ValueError("Missing args['input_file']")

    model = cq.importers.importStep(step_path)

        bb = s.BoundingBox()
        x_min, x_max = bb.xmin, bb.xmax
        y_min, y_max = bb.ymin, bb.ymax
        z_min, z_max = bb.zmin, bb.zmax
        y_mid = 0.5 * (y_min + y_max)

        # Safe X-range for holes (avoid slot region on left and pivot bore region on right)
        safe_x_min = x_slot_end + hole_r + slot_extra_keepout

        # Safe Y-rows (two rows, symmetric about block mid-Y)
        candidate_ys = [y_mid - y_row_offset, y_mid + y_row_offset]

        print(

        # If the safe region is too small, skip modification
