            new_solids.append(s)

        x_pivot = x_max - 5.0

        safe_x_min = x_slot_end + hole_r + slot_extra_keepout

        candidate_ys = [y_mid - y_row_offset, y_mid + y_row_offset]

    return cq.Compound.makeCompound(new_solids)
