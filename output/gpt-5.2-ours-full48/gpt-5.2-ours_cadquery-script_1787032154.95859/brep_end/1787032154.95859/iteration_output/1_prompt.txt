    def safe_normal(face):
        c = face.Center()
        try:
            n = face.normalAt(c)
        except Exception:
            # fallback: try param-space normal
            try:
                n = face.normalAt(0, 0)
            except Exception:
                n = cq.Vector(0, 1, 0)
        ln = (n.Length or 1.0)
        return cq.Vector(n.x / ln, n.y / ln, n.z / ln)

    def edge_hash(e):
        # robust OCC hash
        try:
            return e.wrapped.HashCode(2147483647)
        except Exception:
            try:
                return e.HashCode(2147483647)
            except Exception:
